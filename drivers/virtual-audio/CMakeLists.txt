cmake_minimum_required(VERSION 3.24)
project(mixlive_virtual_audio_transport LANGUAGES CXX)
enable_testing()
add_executable(transport_tests tests.cpp)
target_compile_features(transport_tests PRIVATE cxx_std_17)
if(MSVC)
  target_compile_options(transport_tests PRIVATE /W4 /WX)
else()
  target_compile_options(transport_tests PRIVATE -Wall -Wextra -Werror)
endif()
add_test(NAME transport COMMAND transport_tests)
